[문제 링크](https://www.acmicpc.net/problem/34993) 
## **풀이**
### **문제 요약**
$1$번부터 $N$번까지 $N$개의 점이 있습니다. 어느 세 점도 한 직선 위에 있지 않습니다.

점들의 부분집합을 질문하면 그 볼록 껍질을 이루는 점들의 좌표를 순서 없이 답변받을 수 있습니다. $\lceil \frac{N}{2} \rceil$  회의 질문으로 모든 점의 좌표를 특정하세요.

### **몇 개의 점을 질문해야 최적일까?**
$1$번부터 $N$번 까지 점이 좌표평면에 존재하고 거기에 고무줄을 건다고 생각해 봅시다. 여기서 질문으로 물어보는 **점의 개수**가 바로 고무줄 역할을 합니다. 질문을 통해서 점의 개수를 물어볼 수 있지만 그 점을 기준으로 고무줄로 묶인 다각형의 외부 점만 알 수 있고, 그 다각형 내부에 있을지 모를 점의 위치는 파악할 수 없습니다. 즉, 질문은 점의 개수에 비해 $\lceil \frac{N}{2} \rceil$ 제한되어 있고, 한 번만 많은 점의 위치를 물어보았다간 고무줄에 묶여 내부 점의 위치를 알 수 없고 너무 적게 물어보았다간 질문 개수에 제한됩니다.

만약 한 질문 당 2개의 점을 물어본다면 겹치는 점 하나의 위치만 잡아낼 수 있습니다. 그럼 질문을 100번 해야지만 모든 점을 알 수 있게 되므로 질문 50번 조건에 위배됩니다. 또한 2개의 점을 이으면 선이므로 블록 껍질이 다각형이란 조건에 위배됩니다.

그럼 4개의 점의 위치를 물어보면 어떨까요? 4개의 경우 겹치는 점의 개수가 1개 또는 2개가 나오므로 특정 점의 위치를 잡아낼 수 없습니다. 그러기에 4개의 점은 최적이 아닙니다.

최적의 점의 개수는 **3개**(고무줄이 삼각형)가 됩니다. 3개를 기준으로 잡으면 한 개의 겹치는 점을 알 수 있고 다음 질문에서 다시 겹치는 점 하나를 통해서 나머지 점 하나를 알 수 있게 됩니다. 

예를 들어, ```1 2 3```, ```3 4 5```, ```5 6 7```, $\dots$ 이런 식으로 3개씩 점을 위치를 물어본다면 3이 겹치는 점이고, 다음 질문을 통해 5가 겹치게 됩니다. 3, 5의 위치를 알게 되어 질문 ```3 4 5``` 통해 4의 위치도 특정할 수 있습니다.

즉, 3개의 점을 물어보면 **2개의 점**의 위치를 특정할 수 있게 되어 질문 1번 당 **2개의 점**을 특정하게 됩니다. 그러므로 최대 100개의 점이 있어도 50번의 질문($50 \times 2=100$)을 통해 모든 점의 위치를 특정할 수 있게 되는 것입니다. 그럼 삼각형 내부에 생길 수 있는 점까지 모두 특정할 수 있게 됩니다.

### **$N$이 홀수**
$N$이 홀수라면, ```1 2 3```, ```3 4 5```, ```5 6 7```, $\dots$ , ```N-2 N-1 N```로 나눠볼 수 있고, $\lfloor \frac{N}{2} \rfloor$개의 질문으로 $3$번부터 $N-2$번까지 점을 특정할 수 있습니다. $N$ 홀수이므로 $\lfloor \frac{N}{2} \rfloor$ 통해 1개의 질문이 남게 되고, 남은 하나의 질문으로 ```1 ? N```($5\leq?\leq N-2$)을 통해 $1$번과 $N$번을 특정하고, ```1 2 3```과 ```N-2 N-1 N```을 통해 다시 $2$번과 $N-1$번도 특정해 줍니다.

### **$N$이 짝수**
$N$이 짝수라면 마찬가지로, ```1 2 3```, ```3 4 5```, ```5 6 7```, $\dots$ , ```N-3 N-2 N-1```로 나눠볼 수 있고 $3$번부터 $N-3$번까지 점을 특정할 수 있습니다. 남은 질문 하나로 ```1 N-1 N``` 세 점을 특정하고 이를 통해 다시 $2$번과 $N-2$번도 특정해 줍니다.

import sys
print = lambda *args, sep=" ", end="\n": sys.stdout.write(sep.join(map(str, args)) + end)
input = lambda: sys.stdin.readline().rstrip('\r\n')

def main():
    n = int(input())
    pt = [None]*(n+1)
    qry = [None]*(n+1)
    chk = lambda i, j, idx: (qry[i][j], qry[i][j+1]) == pt[idx]
    for i in range(3, n+1, 2):
        print(f'? 3 {i-2} {i-1} {i}')
        sys.stdout.flush()
        _, *crd = map(int, input().split())
        qry[i] = crd
    for i in range(5, n+1, 2):
        for j in range(0, 6, 2):
            for k in range(0, 6, 2):
                if qry[i-2][j] == qry[i][k] and qry[i-2][j+1] == qry[i][k+1]:
                    pt[i-2] = (qry[i][k], qry[i][k+1])
            if i != 5:
                for j in range(0, 6, 2):
                    for k in range(0, 6, 2):
                        if not chk(i-2, j, i-4) and not chk(i-2, j, i-2):
                            pt[i-3] = (qry[i-2][j], qry[i-2][j+1])
    if n&1:
        print(f'? 3 1 7 {n}')
        sys.stdout.flush()
        _, *crd = map(int, input().split())
        qry[1] = crd
        for j in range(0, 6, 2):
            for k in range(0, 6, 2):
                if qry[1][j] == qry[3][k] and qry[1][j+1] == qry[3][k+1]:
                    pt[1] = (qry[1][j], qry[1][j+1])
        for j in range(0, 6, 2):
            if not chk(3, j, 1) and not chk(3, j, 3):
                pt[2] = (qry[3][j], qry[3][j+1])
        for j in range(0, 6, 2):
            for k in range(0, 6, 2):
                if qry[1][j] == qry[n][k] and qry[1][j+1] == qry[n][k+1]:
                    pt[n] = (qry[1][j], qry[1][j+1])
        for j in range(0, 6, 2):
            if not chk(n, j, n) and not chk(n, j, n-2):
                pt[n-1] = (qry[n][j], qry[n][j+1])
    else:
        print(f'? 3 1 {n-1} {n}')
        sys.stdout.flush()
        _, *crd = map(int, input().split())
        qry[1] = crd
        for j in range(0, 6, 2):
            for k in range(0, 6, 2):
                if qry[1][j] == qry[3][k] and qry[1][j+1] == qry[3][k+1]:
                    pt[1] = (qry[1][j], qry[1][j+1])
        for j in range(0, 6, 2):
            if not chk(3, j, 1) and not chk(3, j, 3):
                pt[2] = (qry[3][j], qry[3][j+1])
        for j in range(0, 6, 2):
            for k in range(0, 6, 2):
                if qry[1][j] == qry[n-1][k] and qry[1][j+1] == qry[n-1][k+1]:
                    pt[n-1] = (qry[1][j], qry[1][j+1])
        for j in range(0, 6, 2):
            if not chk(n-1, j, n-1) and not chk(n-1, j, n-3):
                pt[n-2] = (qry[n-1][j], qry[n-1][j+1])
        for j in range(0, 6, 2):
            if not chk(1, j, 1) and not chk(1, j, n-1):
                pt[n] = (qry[1][j], qry[1][j+1])
    print('!', *[v for i in range(1, n+1) for v in pt[i]])
    sys.stdout.flush()
    return 0
if __name__ == '__main__':
    sys.exit(main())

#include<bits/stdc++.h>
using namespace std;
using ll = long long;
using pii = pair<int, int>;
using ai6 = array<int, 6>;

int main(){
    cin.tie(0); cout.tie(0); ios::sync_with_stdio(false);
    int n, m; cin >> n;
    vector<pii> pt(n+1);
    vector<ai6> qry(n+1);
    auto chk = [&](int i, int j, int idx){
        return qry[i][j] == pt[idx].first && qry[i][j+1] == pt[idx].second;
    };
    for(int i = 3; i <= n; i += 2){
        cout << "? 3 " << i-2 << ' ' << i-1 << ' ' << i << endl;
        cin >> m; for(int j = 0; j < 6; j++) cin >> qry[i][j];
    }
    for(int i = 5; i <= n; i += 2){
        for(int j = 0; j < 6; j += 2){
            for(int k = 0; k < 6; k += 2){
                if(qry[i-2][j] == qry[i][k] && qry[i-2][j+1] == qry[i][k+1]){
                    pt[i-2]= {qry[i][k], qry[i][k+1]};
                }
            }
            if(i != 5){
                for(int j = 0; j < 6; j += 2){
                    for(int k = 0; k < 6; k += 2){
                        if(!chk(i-2, j, i-4) && !chk(i-2, j, i-2)){
                            pt[i-3] = {qry[i-2][j], qry[i-2][j+1]};
                        }
                    }
                }
            }
        }
    }
    if(n&1){
        cout << "? 3 1 5 " << n << endl;
        cin >> m; for(int j = 0; j < 6; j++) cin >> qry[1][j];
        for(int j = 0; j < 6; j += 2){
            for(int k = 0; k < 6; k += 2){
                if(qry[1][j] == qry[3][k] && qry[1][j+1] == qry[3][k+1]){
                    pt[1] = {qry[1][j], qry[1][j+1]};
                }
            }
        }
        for(int j = 0; j < 6; j += 2){
            if(!chk(3, j, 1) && !chk(3, j, 3)){
                pt[2] = {qry[3][j], qry[3][j+1]};
            }
        }
        for(int j = 0; j < 6; j += 2){
            for(int k = 0; k < 6; k += 2){
                if(qry[1][j] == qry[n][k] && qry[1][j+1] == qry[n][k+1]){
                    pt[n] = {qry[1][j], qry[1][j+1]};
                }
            }
        }
        for(int j = 0; j < 6; j += 2){
            if(!chk(n, j, n) && !chk(n, j, n-2)){
                pt[n-1] = {qry[n][j], qry[n][j+1]};
            }
        }
    }
    else{
        cout << "? 3 1 " << n-1 << ' ' << n << endl;
        cin >> m; for(int j = 0; j < 6; j++) cin >> qry[1][j];
        for(int j = 0; j < 6; j += 2){
            for(int k = 0; k < 6; k += 2){
                if(qry[1][j] == qry[3][k] && qry[1][j+1] == qry[3][k+1]){
                    pt[1] = {qry[1][j], qry[1][j+1]};
                }
            }
        }
        for(int j = 0; j < 6; j += 2){
            if(!chk(3, j, 1) && !chk(3, j, 3)){
                pt[2] = {qry[3][j], qry[3][j+1]};
            }
        }
        for(int j = 0; j < 6; j += 2){
            for(int k = 0; k < 6; k += 2){
                if(qry[1][j] == qry[n-1][k] && qry[1][j+1] == qry[n-1][k+1]){
                    pt[n-1] = {qry[1][j],  qry[1][j+1]};
                }
            }
        }
        for(int j = 0; j < 6; j += 2){
            if(!chk(n-1, j, n-1) && !chk(n-1, j, n-3)){
                pt[n-2] = {qry[n-1][j], qry[n-1][j+1]};
            }
        }
        for(int j = 0; j < 6; j += 2){
            if(!chk(1, j, 1) && !chk(1, j, n-1)){
                pt[n] = {qry[1][j], qry[1][j+1]};
            }
        }
    }
    cout << "! "; for(int i = 1; i <= n; i++) cout << pt[i].first << ' ' << pt[i].second << ' '; cout << endl;
    return 0;
}
