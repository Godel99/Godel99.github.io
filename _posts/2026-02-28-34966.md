[문제 링크](https://www.acmicpc.net/problem/34966) 
## **풀이**
### **접근**
목표는 **방의 모든 물건을 제거하는 최소 시행 횟수**를 구하는 것입니다. $M$이 1일 때 시행은 1번이면 족합니다. 

$M$이 2 이상일 때는 시행 횟수에 따른 최단 거리를 구하기 위한 다익스트라와 같은 알고리즘이 필요할테고, 최소 시행 횟수를 저장할 $DP$가 하나 필요해 보입니다. 

### **자명함**
**한 번** 하나의 구멍을 선택하면, 선택된 구멍으로 치울 수 있는 모든 물건을 치우는 게 **최선**입니다. 

두 개의 구멍이 서로 만날 수 없다면, 모든 물건을 치우는 것은 불가능합니다. 두 개의 구멍이 **최소 시행**으로 만나야 한다면, 그 시행은 **두 번 이내**가 될 수 밖에 없습니다. 만약 두 번을 초과한다면, 그것은 최소 시행을 만족하지 못한 겁니다.

### **관점 전환**
그렇다면 최소 시행으로 두 개의 구멍을 먼저 **만나게** 해 봅시다. 두 구멍이 만난 뒤에는 어느 쪽 구멍을 움직이던 모든 물건을 치우는데는 1번의 시행으로 족한 **만능 상태**가 됩니다. 그러므로 두 구멍이 **한 번에** 먼저 만나게 하고, 나머지 한 번으로 남아있는 모든 물건을 치운다면 **최소 두 번 이내의 시행**으로 모든 물건을 치우고 만날 수 있게 됩니다.

처음에 $A$ 에서 출발해 $B$ 의 위치까지 최소 시행을 구하고, $B$ 에서 출발해 $A$의 위치까지 최소 시행을 구한 뒤 둘 중 더 작은 값을 찾아 줍시다.

### **다이얼 알고리즘(Dial's Algorithm)**
**다이얼 알고리즘(Dial's Algorithm)**: 간선 가중치(Weight)가 작고 정수 형태인 그래프에서 다익스트라(Dijkstra) 알고리즘을 최적화한 최단 경로 탐색 알고리즘입니다.

다익스트라로도 충분히 해결가능하지만, 가중치를 두 번 이내의 시행으로 둔다면 가중치가 크지 않습니다. **큐(queue)** 를 가중치로 두면 빠르게 접근이 가능합니다. 가중치가 두 번 이내의 시행이므로 3개의 큐면 충분합니다.

* $dp[y][x]$: $(x, y)$ 좌표까지의 최소 시행 횟수

```python
import sys
print = lambda *args, sep=" ", end="\n": sys.stdout.write(sep.join(map(str, args)) + end)
input = lambda: sys.stdin.readline().rstrip('\r\n')

from collections import deque
from math import inf

dx = (1, -1, 0, 0); dy = (0, 0, 1, -1)

def main():
    n, m = map(int, input().split())
    a = [input() for _ in range(n)]
    start = end = None
    isa = isb = 0
    for i in range(n):
        for j in range(n):
            if a[i][j] == 'A': start = (i, j)
            if a[i][j] == 'B': end = (i, j)
            if a[i][j] == 'a': isa = 1
            if a[i][j] == 'b': isb = 1
    if not isa and not isb: print(0); return 0
    if not isa or not isb: print(1); return 0
    def solved(start, end, type):
        ay, ax = start; by, bx = end
        dq = [deque() for _ in range(3)]
        dp = [[inf]*n for _ in range(n)]
        dq[0].append(start)
        dp[ay][ax] = 0
        res = 0
        while any(dq):
            while not dq[res%3]: res += 1
            while dq[res%3]:
                y, x = dq[res%3].popleft()
                if dp[y][x] < res: continue
                for dir in range(4):
                    ny = y+dy[dir]; nx = x+dx[dir]
                    if ny < 0 or nx < 0 or ny >= n or nx >= n: continue
                    nd = dp[y][x]+(a[ny][nx] == "ab"[type^(res&1)]) 
                    if a[ny][nx] == "ba"[type] and not res: nd = 2
                    if dp[ny][nx] <= nd: continue
                    dp[ny][nx] = nd; dq[nd%3].append((ny, nx))
        if dp[by][bx] > 1: return inf
        return res
    ans = inf; ans = min(ans, solved(start, end, 0), solved(end, start, 1))
    if ans == inf: print(-1)
    else: print(ans)
    return 0
if __name__ == '__main__':
    sys.exit(main())
```

#include<bits/stdc++.h>
using namespace std;
using ll = long long;
using pii = pair<int, int>;
const int INF = 0x3f3f3f3f;
const int dx[] = {1, -1, 0, 0}, dy[] = {0, 0, 1, -1};

int n, m;
vector<vector<char>> a;

int solved(pii start, pii end, bool type){
    auto [ay, ax] = start; auto [by, bx] = end;
    vector<queue<pii>> q(3);
    vector<vector<int>> dp(n, vector<int>(n, INF));
    q[0].push(start);
    dp[ay][ax] = 0;
    int res = 0;
    while(q[0].size() || q[1].size() || q[2].size()){
        while(q[res%3].empty()) res++;
        while(q[res%3].size()){
            auto [y, x] = q[res%3].front(); q[res%3].pop();
            if(dp[y][x] < res) continue;
            for(int dir = 0; dir < 4; dir++){
                int ny = y+dy[dir], nx = x+dx[dir];
                if(ny < 0 || nx < 0 || ny >= n || nx >= n) continue;
                int nd = dp[y][x]+(a[ny][nx] == "ab"[type^(res&1)]);
                if(a[ny][nx] == "ba"[type] && !res) nd = 2;
                if(dp[ny][nx] <= nd) continue; dp[ny][nx] = nd; q[nd%3].push({ny, nx});
            }
        }
    }
    if(dp[by][bx] > 1) return INF;
    return res;
}

int main(){
    cin.tie(0); cout.tie(0); ios::sync_with_stdio(false);
    cin >> n >> m;
    a.assign(n, vector<char>(n, ' '));
    pii start, end; start = end = {};
    bool isa, isb; isa = isb = 0;
    for(int i = 0; i < n; i++) for(int j = 0; j < n; j++){
        cin >> a[i][j];
        if(a[i][j] == 'A') start = {i, j};
        if(a[i][j] == 'B') end = {i, j};
        if(a[i][j] == 'a') isa = 1;
        if(a[i][j] == 'b') isb = 1;
    }
    if(!isa && !isb){cout << 0; return 0;}
    if(!isa || !isb){cout << 1; return 0;}
    int ans = INF; ans = min(ans, solved(start, end, 0)); ans = min(ans, solved(end, start, 1));
    if(ans == INF) cout << -1;
    else cout << ans;
    return 0;
}
