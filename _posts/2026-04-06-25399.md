[문제 링크](https://www.acmicpc.net/problem/25399) 
## **풀이**
정수 $N$을 서로 다른 양의 제곱수들의 합 또는 차로 표현할 때, 사용되는 제곱수의 개수를 **최소**화해야 합니다.

### **분기**
$N<0$일 경우, 마지막에 부호만 바꾸면 되므로 일반성을 유지하기 위해 $N>0$로 바꿔줍니다. 제곱수의 개수를 최소화하기 위해 순서대로 **그리디**하게 분기를 나눠줍니다.

* $N=0,\ 2$경우는 3개의 제곱수를 가지는 경우로 먼저 예외 처리를 해 줍니다.
* $N=m^2$을 만족하면 자명하게 제곱수의 개수는 1개입니다.
* $N=2m+1$이 **홀수**라면, 인접한 두 제곱의 차 $(m+1)^2 - m^2$로 나타낼 수 있고, 그 제곱수의 개수는 2개입니다.
* $N=4m$이 **4의 배수**라면, $(m+1)^2 - (m-1)^2=4m$임을 이용해 두 제곱의 차로 차로 나타낼 수 있고, 그 제곱수의 개수는 홀수와 동일하게 2개입니다.
* 그 외에는 $N$까지 전수조사($N=\sqrt N\times\sqrt N$)를 통해서 **2개**의 제곱수로 나타낼 수 있는지 확인해 줍니다.
* 만약 전수조사로도 나타낼 수 없다면, 이는 **3개**의 양의 제곱수로 $N$을 분할할 수밖에 없다는 뜻이 됩니다. 위의 조건들을 모두 제외했을 때, 남은 경우는 $N=4m+2$만 남습니다. 왜냐면 정수 4로 나눴을 때 나머지는 0, 1, 2, 3이며, 홀수에서 1, 3이 걸러지고 4의 배수에서 0이 걸러집니다. 그래서 $4m+2$의 경우만 남게 되는 것입니다. $4m+2$는 짝수이므로 $\pm1^2$를 더하거나 빼줘서 홀수로 만들어주고, 홀수의 경우를 적용해 $-1^2+(m+1)^2-m^2$와 같이 3개의 양의 제곱수로 나타내 줍니다.

```python
import sys 
print = lambda *args, sep=" ", end="\n": sys.stdout.write(sep.join(map(str, args)) + end)
input = lambda: sys.stdin.readline().rstrip('\r\n')

from math import isqrt

def main():
    n = int(input())
    op = ['-', '+']
    rv = 0 < n
    n = abs(n)
    if n == 0: print('3\n+ 5\n- 4\n- 3\n'); return
    if n == 2: print(f'3\n{op[rv]} 6\n{op[~rv]} 5\n{op[~rv]} 3'); return
    r = isqrt(n)
    if r*r == n: print(f'1\n{op[rv]} {r}'); return
    if n&1: print(f'2\n{op[rv]} {n//2+1}\n{op[~rv]} {n//2}'); return
    if n&3 == 0: print(f'2\n{op[rv]} {n//4+1}\n{op[~rv]} {n//4-1}'); return
    for p in range(1, isqrt(n//2)+1):
        t = n-p*p
        re = isqrt(t)
        if re*re == t: print(f'2\n{op[rv]} {re}\n{op[rv]} {p}'); return
    print(f'3\n{op[~rv]} 1\n{op[rv]} {n//2+1}\n{op[~rv]} {n//2}')
    return  
if __name__ == '__main__':
    main()
```
```c++
#include<bits/stdc++.h>
using namespace std;
using ll = long long;

int main(){
    cin.tie(0); cout.tie(0); ios::sync_with_stdio(false);
    ll n; cin >> n;
    const char *op[] = {"-", "+"};
    bool rv = 0 < n;
    n = abs(n);
    if(n == 0) return !(cout << "3\n+ 5\n- 4\n- 3\n");
    if(n == 2) return !(cout << "3\n" << op[rv] << " 6\n" << op[!rv] << " 5\n" << op[!rv] << " 3");
    if(n == (ll)sqrt(n)*(ll)sqrt(n)) return !(cout << "1\n" << op[rv] << " " << (ll)sqrt(n));
    if(n&1) return !(cout << "2\n" << op[rv] << " " << n/2+1 << '\n' << op[!rv] << " " << n/2);
    if((n&3) == 0) return !(cout << "2\n" << op[rv] << " " << n/4+1 << "\n" << op[!rv] << " " << n/4-1);
    for(ll p = 1; p <= (ll)sqrt(n/2); p++){
        ll t = n-p*p;
        ll re = (ll)sqrt(t);
        if(re*re == t) return !(cout << "2\n" << op[rv] << " " << re << "\n" << op[rv] << " " << p);
    }
    cout << "3\n" << op[!rv] << " 1\n" << op[rv] << " " << n/2+1 << "\n" << op[!rv] << " " << n/2;
    return 0;
}
```
