[문제 링크](https://www.acmicpc.net/problem/7453) 
## **풀이**
정수로 이루어진 크기가 같은 배열 A, B, C, D가 있습니다.

A[a], B[b], C[c], D[d]의 합이 0인 (a, b, c, d) 쌍의 개수를 구해야 합니다.

### **중간에서 만나기**
$A[a]+B[b]+C[c]+D[d]$를 구하는데 굳이 4개를 나눠서 생각하지 말고 2개씩 합쳐서 두 개의 합이 0인 쌍의 개수를 찾아줘도 됩니다. 그럼 복잡도가 $\mathcal{O}(N^4)$에서 $\mathcal{O}(N^2)$까지 줄어듭니다. 이러한 접근을 **중간에서 만나기**라고 합니다.

쌍의 개수를 구해야 하므로, **가능한 최대 범위($SIZE$)** 를 배열로 만들어서 최대 범위에서 **빼기**를 통해 두 정수가 같은지를 판별해 줍니다. 배열로 처리하기 위해 $A[a]+B[b]=-(C[c]+D[d])$식으로 계산해 주면 음수 인덱스를 처리할 수 있습니다. 그러기 위해서 $A[a],\ B[b],\ C[c],\ D[d]$를 먼저 **오름차순으로 정렬**해서 빼기를 통해 그 합의 값이 점점 0에 수렴할 수 있도록 처리합니다.

```python
import sys
input = lambda: sys.stdin.readline().rstrip()
print = lambda *args, sep=" ", end="\n": sys.stdout.write(sep.join(map(str, args)) + end)

def main():
    SIZE = 2**20
    n = int(input())
    A, B, C, D = [0]*n, [0]*n, [0]*n, [0]*n
    for i in range(n): 
        A[i], B[i], C[i], D[i] = map(int, input().split())
        C[i] = -C[i]; D[i] = -D[i]
    A.sort(); B.sort(); C.sort(); D.sort()
    AB = CD = cnt = ans = 0 
    LM = min(A[0]+B[0], C[0]+D[0])+SIZE-1
    NB, ND = [0]*n, [0]*n
    RK, RS = [0]*SIZE, [0]*SIZE
    while AB < n and CD < n:
        cnt += 1
        for i in range(AB, n):
            l = LM - A[i]
            while NB[i] < n:
                s = l-B[NB[i]]
                if s < 0: break
                if RK[s] == cnt: RS[s] += 1
                else: RK[s] = cnt; RS[s] = 1
                NB[i] += 1
                if NB[i] == n: AB += 1
        for i in range(CD, n):
            l = LM-C[i]
            while ND[i] < n:
                s = l-D[ND[i]]
                if s < 0: break
                if RK[s] == cnt: ans += RS[s]
                ND[i] += 1
                if ND[i] == n: CD += 1
        LM += SIZE
    print(ans)
    return
if __name__ == "__main__":
    main()
```

#include <bits/stdc++.h>
using namespace std;
using ll = long long;

int main(){
    cin.tie(0); cout.tie(0); ios::sync_with_stdio(false);
    const int SIZE = 122333;
    int n, AB, CD, cnt, LM; cin >> n;
    vector<int> A(n), B(n), C(n), D(n), NB(n, 0), ND(n, 0);
    for(int i = 0; i < n; i++){
        cin >> A[i] >> B[i] >> C[i] >> D[i];
        C[i] = -C[i], D[i] = -D[i];
    }
    sort(A.begin(), A.end()); sort(B.begin(), B.end()); sort(C.begin(), C.end()); sort(D.begin(), D.end());
    AB = CD = cnt = 0;
    LM = min(A[0]+B[0], C[0]+D[0])+SIZE-1;
    vector<int> RK(SIZE, 0), RS(SIZE, 0);
    ll ans = 0;
    while(AB < n && CD < n){
        cnt++;
        for(int i = AB; i < n; i++){
            int l = LM-A[i];
            while(NB[i] < n){
                int s = l-B[NB[i]];
                if(s < 0) break;
                if(RK[s] == cnt) RS[s]++;
                else RK[s] = cnt, RS[s] = 1;
                NB[i]++;
                if(NB[i] == n) AB++;
            }
        }
        for(int i = CD; i < n; i++){
            int l = LM-C[i];
            while(ND[i] < n){
                int s = l-D[ND[i]];
                if(s < 0) break;
                if(RK[s] == cnt) ans += RS[s];
                ND[i]++;
                if(ND[i] == n) CD++;
            }
        }
        LM += SIZE;
    }
    cout << ans;
    return 0;
}
