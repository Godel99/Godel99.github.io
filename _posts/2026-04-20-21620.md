[문제 링크](https://www.acmicpc.net/problem/21620) 
## **풀이**
구간 $[x_i,\ y_i]$의 $\gcd$가 모두 $z_i$라고 정의하는 질의가 $M$개 주워질 때, 모든 질의를 만족하는 $1\leq A_i \leq 10^9$를 원소로 가지는 $N$개의 수열 $A$를 구해야 합니다. 정답은 여러 개 존재할 수 있고, 만약 불가능할 경우 ```Impossible```을 출력해야 합니다.

### **최대공약소의 최소공배수**
질의를 만족하기 위해 각 원소마다 $\gcd$가 많을수록 만족하는 수열 $A$를 찾기 힘들어집니다. 일단 질의의 각 원소의 $\gcd$를 만족하는 모든 경우를 찾아내서 거기서 다시 $\gcd$를 검사해 줘야 합니다. 만족하는 모든 경우를 찾아주기 위해서는 **$\gcd$의 최소공배수**를 찾아주면 됩니다. $1\leq z_i\leq 16$이므로, 최소공배수는 최대 $5\times7\times9\times11\times13\times16=720,720$로 $10^9$보다 작습니다. 그러니 가능한 최대공약수로부터 모든 최소공배수를 구해 다시 질의의 최대공약수를 검사해 주면, 모든 질의를 만족하는 수열 $A$가 가능한지를 판별해 줄 수 있습니다.

### **세그먼트 트리**
질의 문제에 당골로 나오는 메뉴가 **세그먼트 트리**입니다. 이 문제의 경우 정적 질의가 이루어지기 때문에 세그먼트 트리와 꼭 맞는 문제라고 볼 수는 없지만, 일반적 질의 문제에서 무난한 성능을 내주기 때문에 세그먼트 트리를 씁니다. 최대공약수로부터 가능한 최소공배수를 세그먼트 트리의 리프 노드로 채워주고 그 부모 노드들이 최대공약수를 저장하도록 트리를 설계해 줍니다. 그리고 질의마다 $\gcd$가 가능한지 세그먼트 트리로 검사를 해줍니다. 만약 불가능하다면, ```Impossible``` 출력하고 아니면 이미 구해놓은 최소공배수로 수열 $A$를 만들어주면 됩니다.

```python
import sys
input = lambda: sys.stdin.readline().rstrip()
print = lambda *args, sep=" ", end="\n": sys.stdout.write(sep.join(map(str, args)) + end)

from math import gcd, lcm

def main():
    n, m = map(int, input().split())
    qry = []
    d = [[0]*(n+2) for _ in range(17)]
    for _ in range(m):
        x, y, z =  map(int, input().split())
        qry.append((x, y, z))
        d[z][x] += 1
        d[z][y+1] -= 1
    ans = [1]*n
    for z in range(1, 17):
        cnt = 0
        for i in range(1, n+1):
            cnt += d[z][i]
            if cnt:
                ans[i-1] = lcm(ans[i-1], z)
    tree = [0]*2*n
    for i in range(n): tree[n+i] = ans[i]
    for i in range(n-1, 0, -1): tree[i] = gcd(tree[i<<1], tree[i<<1|1])
    for x, y, z in qry:
        res = 0
        l, r = x+n-1, y+n-1
        while l <= r:
            if l&1: res = gcd(res, tree[l]); l += 1
            if ~r&1: res = gcd(res, tree[r]); r -= 1
            l >>= 1
            r >>= 1
        if res != z: print('Impossible'); return
    print(*ans)
    return
if __name__ == "__main__":
    main()
```

```c++
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ai3 = array<int, 3>;

int main(){
    cin.tie(0); cout.tie(0); ios::sync_with_stdio(false);
    int n, m; cin >> n >> m;
    vector<ai3> qry;
    vector<vector<int>> d(17, vector<int>(n+2, 0));
    while(m--){
        int x, y, z; cin >> x >> y >> z;
        qry.push_back({x, y, z});
        d[z][x]++;
        d[z][y+1]--;
    }
    vector<int> ans(n, 1);
    for(int z = 1; z <= 16; z++){
        int cnt = 0;
        for(int i = 1; i <= n; i++){
            cnt += d[z][i];
            if(cnt) ans[i-1] = lcm(ans[i-1], z);
        }
    }
    vector<int> tree(2*n, 0);
    for(int i = 0; i < n; i++) tree[n+i] = ans[i];
    for(int i = n-1; i > 0; i--) tree[i] = gcd(tree[i<<1], tree[i<<1|1]);
    for(auto [x, y, z] : qry){
        int res = 0, l = x+n-1, r = y+n-1;
        while(l <= r){
            if(l&1) res = gcd(res, tree[l++]);
            if(~r&1) res = gcd(res, tree[r--]);
            l >>= 1;
            r >>= 1;
        }
        if(res != z) return !(cout << "Impossible");
    }
    for(int i : ans) cout << i << ' ';
    return 0;
}
```
