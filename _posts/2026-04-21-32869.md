[문제 링크](https://www.acmicpc.net/problem/32869) 
## **풀이**
* 하나의 뿌리 정점을 가진다. 간선으로 연결된 두 정점은 부모-자식 관계를 가지며, 둘 중 뿌리 정점과 더 가까운 정점이 부모이다.
* 모든 정점은 흰색 또는 검은색으로 칠해져 있다.
  * 자식이 없는 정점은 어떤 색이든 상관없다.
  * 자식이 있는 정점은 모든 자손 정점의 색 중 더 많은 색으로 칠해져야 한다. 즉, 흰색 자손이 더 많다면 흰색, 검은색 자손이 더 많다면 검은색이어야 한다. 흰색 자손과 검은색 자손의 수가 같다면 어떤 색이든 상관없다. 이때, 임의의 서로 다른 두 정점 $a$, $b$에 대해 $a$에서 뿌리 정점으로 가는 단순 경로 사이에 $b$가 있다면 $a$는 $b$의 자손이다.

위 조건을 만족할 때 Even 트리라고 합니다. $N$개의 정점으로 이뤄진 트리가 주어질 때, 각 정점을 기준으로 Even트리를 만족하는 개수를 구하고 오름차순으로 그 정점을 출력해야 합니다.

### **판별**
Even인지 판별을 하기 위해 트리 구조의 DP를 설계해 줘야 합니다. 안 그러면 매번 판별을 위해서 트리 전체 구조를 왕복해야 하므로 시간이 오래 걸립니다. 일단 Even 트리인지 판별하기 위한 조건입니다.

* 리프 노드는 Even입니다. 자기 자신을 제외한 자식 노드가 없기 때문에 항상 Even입니다.
* 루트 노드의 색이 서브 트리의 다수 색이 아니면, Even이 아닙니다.
* Even이 아닌 노드가 서브 트리 안에 있다면, Even이 아닙니다.

적당히 1번 정점으로 시작해서 **DFS**로 트리 구조를 돌면서 Even인지를 판별해 줍니다. 트리 DP를 설계하기 위한 배열을 아래와 같이 선언해 줍니다.

* $isEven[u]$: 정점 $u$를 루트로 하는 트리가 Even을 만족하는가?
* $w[u],\ b[u]$: 정점 $u$를 부모로 하는 서브 트리의 흰색/ 검은색 수
* $cnt[u]$: Even하지 않는 자식의 개수

### **리루팅(Re-rooting)**
모든 정점을 한 번씩 루트로 끌어올려야 합니다. 이러한 작업을 리루팅이라고 합니다. 두 정점이 서로 교환될 때, 두 정점의 자식 정보는 바뀌지 않고 두 정점의 부모 자식 관계만 바뀝니다. 따라서 한 번의 DFS로 설계된 DP에 Even인지를 판별할 **정보**를 기록하고, 기록을 바탕으로 판별만 한 번 더 해주면 됩니다.

리루팅 시에 한 가지 주의할 점은 어느 서브 트리에 Even이 아닌 자식이 있다면, 해당 자식을 루트로 끌어올리지 않으면 여전히 서브 트리는 Even이 아닙니다. 만약 서브  트리에 **Even이 아닌 자식이 2개 이상** 존재한다면, 그 서브 트리는 결코 Even이 될 수 없습니다. 이러한 조건을 통해 **가지치기**를 한다면, 좀 더 빠르게 문제를 해결할 수 있습니다.

```python
import sys
input = lambda: sys.stdin.readline().rstrip()
print = lambda *args, sep=" ", end="\n": sys.stdout.write(sep.join(map(str, args)) + end)

sys.setrecursionlimit(300000)

def main():
    n = int(input())
    e = [[] for _ in range(n+1)]
    c = [0]
    tw = tb = 0
    isEven = [0]*(n+1)
    w, b, cnt = [0]*(n+1), [0]*(n+1), [0]*(n+1)
    ans = []
    for i in [*map(int, input().split())]:
        c.append(i)
        if i: tb += 1
        else: tw += 1
    for _ in range(n-1):
        u, v = map(int, input().split())
        e[u].append(v)
        e[v].append(u)
    def dfs(cur, pre):
        isEven[cur] = 1
        if c[cur]: b[cur] = 1
        else: w[cur] = 1
        for nxt in e[cur]:
            if pre == nxt: continue
            dfs(nxt, cur)
            b[cur] += b[nxt]
            w[cur] += w[nxt]
            isEven[cur] &= isEven[nxt]
            if not isEven[nxt]: cnt[cur] += 1
        if c[cur] and (w[cur] >= b[cur]) or not c[cur] and (b[cur] >= w[cur]): isEven[cur] = 0
    def sol(cur, pre):
        if not cnt[cur] and (c[cur] and (tw < tb) or not c[cur] and (tb < tw)): ans.append(cur)
        if cnt[cur] >= 2: return
        for nxt in e[cur]:
            if nxt == pre: continue
            if cnt[cur] and isEven[nxt]: continue
            if c[cur] and (tw-w[nxt] >= tb-b[nxt]) or not c[cur] and (tb-b[nxt] >= tw-w[nxt]): continue
            sol(nxt, cur)
    dfs(1, 1)
    sol(1, 1)
    ans.sort()
    print(len(ans))
    print(*ans)
    return
if __name__ == "__main__":
    main()
```

#include <bits/stdc++.h>
using namespace std;
using ll = long long;

int main(){
    cin.tie(0); cout.tie(0); ios::sync_with_stdio(false);
    int n; cin >> n;
    vector<vector<int>> e(n+1);
    vector<int> c(n+1, 0), isEven(n+1, 0), w(n+1, 0), b(n+1, 0), cnt(n+1, 0), ans;
    int tw = 0, tb = 0;
    for(int i = 1; i <= n; i++){
        cin >> c[i];
        if(c[i]) tb++;
        else tw++;
    }
    for(int i = 0; i < n-1; i++){
        int u, v; cin >> u >> v;
        e[u].push_back(v);
        e[v].push_back(u);
    }
    function<void(int, int)> dfs = [&](int cur, int pre){
        isEven[cur] = 1;
        if(c[cur]) b[cur] = 1;
        else w[cur] = 1;
        for(int nxt : e[cur]){
            if(pre == nxt) continue;
            dfs(nxt, cur);
            b[cur] += b[nxt];
            w[cur] += w[nxt];
            isEven[cur] &= isEven[nxt];
            if(!isEven[nxt]) cnt[cur]++;
        }
        if(c[cur] && (w[cur] >= b[cur]) || !c[cur] && (b[cur] >= w[cur])) isEven[cur] = 0;
    };
    function<void(int, int)> sol = [&](int cur, int pre){
        if(!cnt[cur] && (c[cur] && (tw < tb) || !c[cur] && (tb < tw))) ans.push_back(cur);
        if(cnt[cur] >= 2) return;
        for(int nxt : e[cur]){
            if(nxt == pre) continue;
            if(cnt[cur] && isEven[nxt]) continue;
            if(c[cur] && (tw-w[nxt] >= tb-b[nxt]) || !c[cur] && (tb-b[nxt] >= tw-w[nxt])) continue;
            sol(nxt, cur);
        }
    };
    dfs(1, 1);
    sol(1, 1);
    sort(ans.begin(), ans.end());
    cout << ans.size() << '\n';
    for(auto i : ans) cout << i << ' ';
    return 0;
}
