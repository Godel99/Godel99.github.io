[문제 링크](https://www.acmicpc.net/problem/14500)
## **풀이**
임의의 수로 구성된 2차원 배열에 테트로미노라고 불리는 모양의 블록을 하나 놓았을 때 그 블록이 포함하는 배열의 합의 최댓값을 구하는 문제입니다. 총 5가지의 모양이 있는데 **브루트 포스** 방식으로 모양을 바꿔가며 또 회전해가며 일일이 최댓값을 찾아갈 수도 있습니다. 그러나, 모양은 크게 중요하지 않고 구하고자 하는 것이 최댓값이므로 좀 더 나은 접근이 있습니다.

### **DFS, 백트래킹 + 붓 그리기**
'**ㅜ**'모양을 제외하고는 붓으로 그림을 그리듯 **한 점에서 연속해서 모양**을 만들어 낼 수 있다는 특징이 있습니다. 예를 들어 '**ㄴ, ㅡ, ㅁ, ㄹ**'와 같은 모양은 한 점에서 시작해 되돌아가지 않고 연속해서 모양을 만들 수 있습니다. 이 점을 착안해 **DFS** 이용한다면 '**ㅜ**' 제외한 모든 모양의 움직임을 따라갈 수 있게 됩니다. 

각 모양은 모두 **4칸**으로 구성되어 있으므로, 4번에 DFS로 상하좌우를 방문해주면 'ㅜ'모양을 제외한 모양은 모두 처리가 가능합니다.

### **'ㅜ' 처리**
'ㅜ'의 경우는 나머지 모양과 다르게, 연속해서 움직여선 모양을 만들 수 없습니다. 모양을 보면 중앙을 기준으로 오른쪽을 찍고 다시 중앙으로 돌아와 아래쪽을 확인해야 하는 구조입니다. 따라서 중앙인 2번째까지 가서 3번째를 방문 후 다시 2번째에서 4번째 칸을 방문하는 겁니다. 따라서 다른 모양의 DFS 동작 중간 깊이가 **2일 때** 'ㅜ'모양만 따로 처리해 주면 편합니다.

```python
import sys 
print = lambda *args, sep=" ", end="\n": sys.stdout.write(sep.join(map(str, args)) + end)
input = lambda: sys.stdin.readline().rstrip('\r\n')

dx, dy = (1, -1, 0, 0), (0, 0, 1, -1)
sys.setrecursionlimit(2000)

def main():
    n, m = map(int, input().split())
    board = [list(map(int, input().split())) for _ in range(n)]
    visited = [[0]*m for _ in range(n)]
    ans = 0
    maxv = max(map(max, board))
    def dfs(x, y, d, s):
        nonlocal ans
        if s+maxv*(4-d) <= ans: return
        if d == 4: ans = max(ans, s); return
        for dir in range(4):
            nx, ny = x+dx[dir], y+dy[dir]
            if 0 > nx or nx >= n or 0 > ny or ny >= m or visited[nx][ny]: continue
            if d == 2:
                visited[nx][ny] = 1
                dfs(x, y, d+1, s+board[nx][ny])
                visited[nx][ny] = 0
            visited[nx][ny] = 1
            dfs(nx, ny, d+1, s+board[nx][ny])
            visited[nx][ny] = 0
    for i in range(n):
        for j in range(m):
            visited[i][j] = 1
            dfs(i, j, 1, board[i][j])
            visited[i][j] = 0
    print(ans)
if __name__ == '__main__':
    main()
```

#include<bits/stdc++.h>
using namespace std;
using ll = long long;

const int dx[] = {1, -1, 0, 0}, dy[] = {0, 0, 1, -1};
vector<vector<int>> board;
vector<vector<bool>> visited;
int n, m, ans, maxv;

void dfs(int x, int y, int d, int s){
    if(s+maxv*(4-d) <= ans) return;
    if(d == 4){ans = max(ans, s); return;}
    for(int dir = 0; dir < 4; dir++){
        int nx = x+dx[dir], ny = y+dy[dir];
        if(0 > nx || nx >= n || 0 > ny || ny >= m || visited[nx][ny]) continue;
        if(d == 2){
            visited[nx][ny] = 1;
            dfs(x, y, d+1, s+board[nx][ny]);
            visited[nx][ny] = 0;
        }
        visited[nx][ny] = 1;
        dfs(nx, ny, d+1, s+board[nx][ny]);
        visited[nx][ny] = 0;
    }
}

int main(){
    cin.tie(0); cout.tie(0); ios::sync_with_stdio(false);
    cin >> n >> m;
    board.assign(n, vector<int>(m));
    visited.assign(n, vector<bool>(m, 0));
    for(int i = 0; i < n; i++) for(int j = 0; j < m; j++) cin >> board[i][j];
    for(int i = 0; i < n; i++) for(int j = 0; j < m; j++) maxv = max(maxv, board[i][j]);
    for(int i = 0; i < n; i++) for(int j = 0; j < m; j++){
        visited[i][j] = 1;
        dfs(i, j, 1, board[i][j]);
        visited[i][j] = 0;
    }
    cout << ans;
    return 0;
}
