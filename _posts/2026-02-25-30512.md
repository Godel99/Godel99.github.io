[문제 링크](https://www.acmicpc.net/problem/30512) 
## **풀이**
### **관점 전환**
수열에서 특정 범위의 $min(A_i,\ x)$ 연산을 적용하는 건 어렵지 않습니다. 문제는 업데이트 후 **잊힌 원소**의 개수를 구하는 것이 까다롭습니다. 잊힌 원소는 업데이트 직후뿐만 아니라 모든 업데이트 이후까지 생각해야 하기 때문에 **연산의 시점**을 저장할 새로운 접근 방법이 필요합니다.

### **느리게 갱신되는(lazy propagation) 세그먼트 트리**
보통 세그먼트 트리는 구간의 합을 구하거나 그 최소 최대의 값을 구할 때 쓰이는 알고리즘이지만 이 문제에선 연산이 이뤄지는 시점이 중요하므로, 각 연산을 저장하고 그 연산이 이뤄지는 시점을 저장하기 위해 **느리게 갱신되는 세그먼트 트리**를 이용해 주는 겁니다.

어떤 구간이 변경될 때마다, 세그먼트 트리의 중간 $seg$ 마다 그 명령을 저장해 주고, $lazy$ 에는 마지막 시점을 저장해 두는 겁니다. 마지막 시점이란 업데이트 이후 더 이상 업데이트가 없는 시점을 의미합니다.

모든 업데이트 $q$ 가 끝난 뒤엔 $INF$ 값을 세그먼트 트리의 모든 노드에 적용해 그동안 **미뤄뒀던** 업데이트와 마지막 시점을 리프 노드까지 전파해 줍시다. 가장 큰 $INF$ 값을 적용해야 $min$ 연산에서 각 노드의 값이 변경되지 않고 업데이트만 진행될 수 있습니다.

마지막 시점을 **누적한** 합을 구해주면 잊힌 원소의 개수를 구할 수 있습니다.

```python
import sys
print = lambda *args, sep=" ", end="\n": sys.stdout.write(sep.join(map(str, args)) + end)
input = lambda: sys.stdin.readline().rstrip('\r\n')

from math import inf

def main():
    n = int(input())
    A = [0]+list(map(int, input().split()))
    T = [0]*(n+1)
    q = int(input())
    ans = [0]*(q+1)
    seg = [inf]*4*n
    lazy = [0]*4*n  
    def update(now, l, r, L, R, x, t):
        if R < l or r < L: return
        if L <= l and r <= R:
            if seg[now] > x:
                seg[now] = x
                lazy[now] = max(lazy[now], t)
            A[L] = seg[now]
            T[L] = lazy[now]
            return
        if seg[now<<1] > seg[now]:
            seg[now<<1] = seg[now]
            lazy[now<<1] = max(lazy[now<<1], lazy[now])
        if seg[now<<1|1] > seg[now]:
            seg[now<<1|1] = seg[now]
            lazy[now<<1|1] = max(lazy[now<<1|1], lazy[now])
        seg[now] = inf
        lazy[now] = 0
        mid = (l+r)>>1
        update(now<<1, l, mid, L, R, x, t)
        update(now<<1|1, mid+1, r, L, R, x, t)
    for i in range(1, n+1): update(1, 1, n, i, i, A[i], 0)
    for i in range(1, q+1):
        l, r, x = map(int, input().split())
        update(1, 1, n, l, r, x, i)
    for i in range(1, n+1):
        update(1, 1, n, i, i, inf, 0)
        print(A[i], end=' ')
        ans[T[i]] += 1
    print()
    for i in range(1, q+1): 
        ans[i] += ans[i-1]
        print(ans[i], end=' ')
    return 0
if __name__ == '__main__':
    sys.exit(main())
```
```c++
#include<bits/stdc++.h>
using namespace std;
using ll = long long;
const int INF = 0x3f3f3f3f;

vector<int> A, T, seg, lazy, ans;

void update(int now, int l, int r, int L, int R, int x, int t){
    if(R < l || r < L) return;
    if(L <= l && r <= R){
        if(seg[now] > x){
            seg[now] = x;
            lazy[now] = max(lazy[now], t);
        }
        A[L] = seg[now];
        T[L] = lazy[now];
        return;
    }
    if(seg[now<<1] > seg[now]){
        seg[now<<1] = seg[now];
        lazy[now<<1] = max(lazy[now<<1], lazy[now]);
    }
    if(seg[now<<1|1] > seg[now]){
        seg[now<<1|1] = seg[now];
        lazy[now<<1|1] = max(lazy[now<<1|1], lazy[now]);
    }
    seg[now] = INF;
    lazy[now] = 0;
    int mid = (l+r)>>1;
    update(now<<1, l, mid, L, R, x, t);
    update(now<<1|1, mid+1, r, L, R, x, t);
}

int main(){
    cin.tie(0);cout.tie(0);ios::sync_with_stdio(0);
    int n, q; cin >> n;
    A.assign(n+1, 0);
    T.assign(n+1, 0);
    seg.assign(4*n, INF);
    lazy.assign(4*n, 0);
    for(int i = 1; i <= n; i++){
        cin >> A[i];
        update(1, 1, n, i, i, A[i], 0);        
    }
    cin >> q;
    for(int i = 1; i <= q; i++){
        int l, r, x; cin >> l >> r >> x;
        update(1, 1, n, l, r, x, i);
    }
    ans.assign(q+1, 0);
    for(int i = 1; i <= n; i++){
        update(1, 1, n, i, i, INF, 0);
        cout << A[i] << ' ';
        ans[T[i]]++;
    }
    cout << '\n';
    for(int i = 1; i <= q; i++){
        ans[i] += ans[i-1];
        cout << ans[i] << ' ';
    }
    return 0;
}
```
