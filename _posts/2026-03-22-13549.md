[문제 링크](https://www.acmicpc.net/problem/13549) 
## **풀이**
1차원 배열 위치 $N$에서 세 가지 이동 $x-1$, $x+1$, $x\times 2$ 만으로 위치 $K$까지 최소 몇 초 만에 가는지 구해야 합니다. 특히 기존 숨바꼭질 문제와 다르게 순간 이동 $x\times 2$의 비용이 0초라는 점을 주의 깊게 봐야 합니다.

### **0-1 BFS**
순간 이동의 비용이 0초이기 때문에 기존 이동들보다 우선순위가 높기에 먼저 처리해 줘야 합니다. 이 경우 큐(queue)보다는 덱(deque)을 이용해 비용이 0인 것은 앞에 넣어서 앞에 것을 먼저 처리하고 비용이 1인 것은 뒤에 넣어줍니다.

**주의**해야 할 점은 이동 $x-1$, $x+1$의 순서도 중요하다는 점입니다. 순간 이동이 앞으로만 이동하므로 **$x-1$을 먼저 처리**해줘야 순간 이동의 움직임을 먼저 덱에 넣어서 처리할 수 있게 됩니다. 

### **재귀+그리디**
기존 **0-1 BFS**방식보다 **재귀+그리디**방식도 존재합니다. 순간 이동이 $x\times 2$ 이동하므로 $n$를 2로 분할해가며 $k$를 찾아간다면 기존 **0-1 BFS**의 $\mathcal{O}(K)$복잡도보다 $\mathcal{O}(\log K)$만큼 빠르게 구현이 가능합니다.

```python
import sys 
print = lambda *args, sep=" ", end="\n": sys.stdout.write(sep.join(map(str, args)) + end)
input = lambda: sys.stdin.readline().rstrip('\r\n')

from collections import deque

def main():
    n, k = map(int, input().split())
    if n == k: print(0); return
    maxl = max(n, k)+2
    dp = [-1]*maxl
    dp[n] = 0
    dq = deque([n])
    while dq:
        cur = dq.popleft()
        nxt = cur*2
        if 0 <= nxt < maxl and dp[nxt] == -1:
            dp[nxt] = dp[cur]
            dq.appendleft(nxt)
        for nxt in [cur-1, cur+1]:
            if 0 <= nxt < maxl and dp[nxt] == -1:
                dp[nxt] = dp[cur]+1
                dq.append(nxt)
    print(dp[k])
    return
if __name__ == '__main__':
    main()
```

```c++
int main(){
    cin.tie(0); cout.tie(0); ios::sync_with_stdio(false);

    if(n ==  k) return !(cout << 0);
    int maxl = max(n, k)+2;
    vector<int> dp(maxl, -1);
    dp[n] = 0;
    deque<int> dq; dq.push_back(n);
    while(dq.size()){
        int cur = dq.front(); dq.pop_front();
        if(cur == k) break;
        int nxt = cur*2;
        if(0 <= nxt && nxt < maxl && dp[nxt] == -1){
            dp[nxt] = dp[cur];
            dq.push_front(nxt);
        }
        for(auto nxt: {cur-1, cur+1}){
             if(0 <= nxt && nxt < maxl && dp[nxt] == -1){
                dp[nxt] = dp[cur]+1;
                dq.push_back(nxt);
             }
        }
    }
    cout << dp[k];
    return 0;
}
```

```python
import sys 
print = lambda *args, sep=" ", end="\n": sys.stdout.write(sep.join(map(str, args)) + end)
input = lambda: sys.stdin.readline().rstrip('\r\n')

def main():
    n, k = map(int, input().split())
    def f(n, k):
        if n >= k: return n-k
        if k == 1: return abs(n-1)
        if k&1: return -~min(f(n, k-1), f(n, k+1))
        else: return min(k-n, f(n, k>>1))
    print(f(n, k))
    return
if __name__ == '__main__':
    main()
```
