[문제 링크](https://www.acmicpc.net/problem/31003) 
## **풀이**
### 접근
두 인접한 원소가 서로소에 한해서 자리를 바꿀 수 있습니다. 서로소에 한해서는 버블 정렬을 떠올릴 수 있겠네요. 서로소인 수를 먼저 사전 순으로 정렬하고 그 다음이 서로소가 아닌 수를 정렬하라는 문제입니다. 

서로소인지 판별하는 최대공약수 함수가 하나 필요하겠네요. 서로소가 아닌 수들 간에 간선으로 연결해서 **위상 정렬** 모양을 만들어 주고 서로소인 수의 정렬 뒤로 밀어주는 식으로 구현하면 될 겁니다.

서로소인 수들끼리는 사전 순이므로 **우선순위 큐**를 이용해서 정렬해 줍시다.

### 위상 정렬
서로소가 아닌 수에 대한 간선을 만들어주고 위상 정렬을 구현합시다. 진입 차수가 낮은 것부터 차례대로 정렬을 해주면 알아서 서로소가 아닌 수들은 뒤로 밀려서 정렬을 하게 됩니다.

### 우선순위 큐
서로소인 수들은 오름차순 정렬이 필요하고 진입 차수가 0인 것을 먼저 처리해 줘야 하므로 **우선순위 큐**를 이용해 정렬을 해줍시다.

```python
import sys
def print(*args, sep=" ", end="\n"): return sys.stdout.write(sep.join(map(str, args)) + end)
def input(): return sys.stdin.readline().rstrip()

import heapq
from math import gcd

def main():
    n = int(input())
    a = tuple(map(int, input().split()))
    e = [[] for _ in range(n)]
    deg = [0]*n
    for i in range(n):
        for j in range(i):
            if gcd(a[j], a[i]) != 1: 
                e[j].append(i)
                deg[i] += 1
    hq = []
    for i in range(n):
        if deg[i] == 0: heapq.heappush(hq, (a[i], i))
    while hq:
        x, cur = heapq.heappop(hq)
        print(x, end=' ')
        for nxt in e[cur]:
            deg[nxt] -= 1
            if deg[nxt] == 0: heapq.heappush(hq, (a[nxt], nxt))
    return 0
if __name__ == '__main__':
    sys.exit(main())
```
```c++
#include<bits/stdc++.h>
using namespace std;
using ll = long long;
using pii = pair<int, int>;
using min_heap = priority_queue<pii, vector<pii>, greater<pii>>;

int main(){
    cin.tie(0); cout.tie(0); ios::sync_with_stdio(false);
    int n; cin >> n;
    vector<vector<int>> e(n);
    vector<int> deg(n, 0), a(n);
    for(int i = 0; i < n; i++){
        cin >> a[i];
        for(int j = 0; j < i; j++){
            if(gcd(a[j], a[i]) != 1){
                e[j].push_back(i);
                deg[i]++;
            }
        }
    }
    min_heap pq;
    for(int i = 0; i < n; i++) if(!deg[i]) pq.push({a[i], i});
    while(pq.size()){
        auto [x, cur] = pq.top(); pq.pop();
        cout << x << ' ';
        for(auto& nxt : e[cur]){
            if(!(--deg[nxt])) pq.push({a[nxt], nxt});
        }
    }
    return 0;
}
```

* 공간복잡도: $\mathcal{O}(N^2)$
