[문제 링크](https://www.acmicpc.net/problem/34992) 
## **풀이**
$0$부터 $2N$까지의 수로 이루어진 이진 트리에서 0을 포함한 짝수는 리프 노드가 되며, 홀수는 부모 노드가 되도록 이진 트리를 만듭니다.

이 트리에서 짝수를 탐색했을 때, 짝수로만 이루어진 집합 $S$의 원소는 찾을 수 있도록 하고 $S$의 원소가 아닌 짝수는 찾을 수 없도록 이진 트리를 설계해야 합니다.

$0$부터 $2N$까지의 수로 이루어진 이진 트리이므로 $0$과 $2N$은 반드시 찾아집니다. 다만, 입력으로 $0$이 없는 경우만 예외 처리를 해주면 됩니다.

### **트리 설계**
<p align="center"/><img src="/assets/img/posts/34992.png" alt="편향 이진 트리" width="500">

짝수를 오름차순으로 왼쪽부터 오른쪽으로 쭉 깔아두고 부모 노드를 이진 트리에 맞게 계단식으로 **편향된 트리**를 만든다면, 모든 짝수를 찾을 수 있습니다. 이러한 트리에서 특정 구간의 짝수를 찾지 못하게 하려면, 홀수로 이루어진 부모의 위치를 왼쪽부터 순서대로 **서로 바꿔준다면** 특정 구간의 짝수를 찾지 못하도록 이진 트리를 설계할 수 있습니다.

```python
import sys 
print = lambda *args, sep=" ", end="\n": sys.stdout.write(sep.join(map(str, args)) + end)
input = lambda: sys.stdin.readline().rstrip('\r\n')

def main():
    k = int(input())
    s = [0]+list(map(int, input().split()))
    n = s[-1]
    par = [0]*(n//2+1)
    L = [0]*(n+1); R = [0]*(n+1)
    if s[1] != 0: print(-1); return
    print(n//2)
    for i in range(1, n+1, 2): par[i//2+1] = i
    idx = 2
    for i in range(1, n//2):
        if s[idx] != i*2: par[i], par[i+1] = par[i+1], par[i]
        else: idx += 1
    for i in range(1, n//2+1): L[par[i]], R[par[i]] = par[i-1], i*2
    for i in range(1, n+1, 2): print(L[i], R[i])
if __name__ == '__main__':
    main()
```
```c++
#include<bits/stdc++.h>
using namespace std;
using ll = long long;
using pii = pair<int, int>;

int main(){
    cin.tie(0); cout.tie(0); ios::sync_with_stdio(false);
    int k; cin >> k;
    vector<int> s(k+1);
    for(int i = 1; i <= k; i++) cin >> s[i];
    if(s[1] != 0){cout << -1; return 0;}
    int n = s[k];
    vector<int> par(n/2+1, 0);
    vector<pii> tree(n+1, {0, 0});
    cout << n/2 << '\n';
    for(int i = 1; i <= n; i += 2) par[i/2+1] = i;
    int idx = 2;
    for(int i = 1; i < n/2; i++){
        if(s[idx] != i*2) swap(par[i], par[i+1]);
        else idx++;
    }
    for(int i = 1; i <= n/2; i++) tree[par[i]] = {par[i-1], i*2};
    for(int i = 1; i <= n; i += 2) cout << tree[i].first << ' ' << tree[i].second << '\n';
    return 0;
}
```
