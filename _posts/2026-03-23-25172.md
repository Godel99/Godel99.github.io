[문제 링크](https://www.acmicpc.net/problem/25172) 
## **풀이**
그래프 문제에 일종이고, 분리 집합의 냄새가 납니다. 문제에서 요구하는 이상적 경로는 단일 경로입니다. 그래프에서 노드를 추가하거나 빼거나 할 때마다 단일 경로인지를 확인하는 것은 복잡하므로 이를 단순화하는 게 좋습니다.

### **오프라인 쿼리**
일종의 오프라인 쿼리 방법으로, **모든 질의를 한 번에 다 받아놓고, 유리한 순서로 재배치해서 해결하는 방법**을 오프라인 쿼리라고 합니다. 그래프를 다 그려놓고 하나씩 노드를 삭제해 가는 게 아니라 입력을 모두 받아놓고 거꾸로 노드를 하나씩 추가해가며 그래프를 그려 갑니다.

주의해야 할 점은 결과를 출력할 때 **거꾸로** 출력을 해줘야 한다는 것입니다. 스택을 이용해도 좋고 배열을 거꾸로 출력해줘도 좋습니다.

### **분리 집합(Union by Size)**
그래프의 노드가 하나씩 추가될 때마다 간선으로 연결되어 있다면 같은 집합으로 묶어줘서 단일 경로가 형성되는지 확인해 줍니다. 이럴 때 **크기별** 분리 집합을 쓰면 유용합니다.

```python
import sys 
print = lambda *args, sep=" ", end="\n": sys.stdout.write(sep.join(map(str, args)) + end)
input = lambda: sys.stdin.readline().rstrip('\r\n')

def main():
    n, m = map(int, input().split())
    e = [[] for _ in range(n+1)]
    vis = [0]*(n+1)
    par = [-1]*(n+1)
    def f(x):
        if par[x] < 0: return x
        par[x] = f(par[x])
        return par[x]
    def u(x, y):
        x, y = f(x), f(y)
        if x == y: return
        if par[x] > par[y]: x, y = y, x
        par[x] += par[y]
        par[y] = x
    for _ in range(m):
        x, y = map(int, input().split())
        e[x].append(y)
        e[y].append(x)
    st = [int(input()) for _ in range(n)]
    ans = []
    for i in range(1, n+1):
        cur = st[-i]
        vis[cur] = 1
        for nxt in e[cur]:
            if vis[nxt]: u(cur, nxt)
        if par[f(cur)] == -i: ans.append('CONNECT\n')
        else: ans.append('DISCONNECT\n')
    ans.reverse()
    ans.append('DISCONNECT\n')
    print(''.join(ans))
    return
if __name__ == '__main__':
    main()
```
```c++
#include<bits/stdc++.h>
using namespace std;
using ll = long long;

vector<int> par;

int f(int x){
    if(par[x] < 0) return x;
    par[x] = f(par[x]);
    return par[x];
}

void u(int x, int y){
    x = f(x), y = f(y);
    if(x == y) return;
    if(par[x] > par[y]) swap(x, y);
    par[x] += par[y];
    par[y] = x;
}

int main(){
    cin.tie(0); cout.tie(0); ios::sync_with_stdio(false);
    int n, m; cin >> n >> m;
    vector<vector<int>> e(n+1);
    for(int i = 0; i < m; i++){
        int x, y; cin >> x >> y;
        e[x].push_back(y);
        e[y].push_back(x);
    }
    vector<int> st(n);
    vector<bool> vis(n+1, 0);
    par.assign(n+1, -1);
    for(int i = 0; i < n; i++){
        cin >> st[i];
    }
    vector<bool> ans(n);
    for(int i = 1; i <= n; i++){
        int cur = st[n-i];
        vis[cur] = 1;
        for(int nxt: e[cur]) if(vis[nxt]) u(cur, nxt);
        ans[i-1] = par[f(cur)] == -i;
    }
    for(int i = n-1; i >= 0; i--) cout << (ans[i] ? "CONNECT\n" : "DISCONNECT\n");
    cout << "DISCONNECT\n";
    return 0;
}
```
