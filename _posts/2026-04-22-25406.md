[문제 링크](https://www.acmicpc.net/problem/25406)
## **풀이**
$N$개의 수열 $A_1, \dots , A_N$에서 $A_{P_i} ≠ A_{P_{i+1}}$를 만족하는 순열 $P$ 중 사전 순으로 가장 앞선 순열 하나를 구해야 합니다. 만족하는 순열이 없을 경우 $-1$을 출력해야 합니다.

### **그리디**
예를 들어 수열에 $A_1$원소가 굉장히 많다고 하면, 연속해서 $A_1$을 가져갈 수 없으므로 $A_1,\ A_2,\ A_1,\ A_3,\ A_1,\cdots$와 같이 $A_1$를 번갈아가며 선택해야 합니다. 따라서, $A_1$를 최대한 선택할 수 있는 개수는 $\frac{N}{2}+1$보다 크면 안 됩니다. 만약 더 크다면 $N$개를 넘어서는 순열이 나와 만족하는 정답이 나올 수 없으므로, $-1$을 출력해 줍니다.

그러니 수열의 각 원소의 개수를 미리 파악해 두는 것이 중요합니다. 처음에는 사전 순으로 원소를 선택해 가다가, 선택한 원소의 개수가 원소의 개수 중에 최댓값과 같아지면, 최댓값의 원소를 기준으로 번갈아가며 다른 원소를 끼어 넣어주며 선택해 가면 됩니다.

구현은 배열을 이용한 이중 연결 리스트를 이용했습니다. 중간에 선택된 원소를 빼야 할 때 쓰기 좋은 자료구조가 이중 연결 리스트입니다.

```python
import sys
input = lambda: sys.stdin.readline().rstrip()
print = lambda *args, sep=" ", end="\n": sys.stdout.write(sep.join(map(str, args)) + end)

def main():
    n = int(input())
    a = [*map(int, input().split())]
    ans, cnt, head, nxt, pre = [0]*n, [0]*(n+1), [-1]*(n+1), [-1]*(n+1), [-1]*(n+1)
    vis = bytearray(n)
    mcnt = 0
    for i in range(n): cnt[a[i]] += 1
    def Add(c, x):
        nxt[x] = head[c]
        if head[c] != -1: pre[head[c]] = x
        head[c] = x
    def Remove(c, x):
        nonlocal mcnt
        if pre[x] != -1: nxt[pre[x]] = nxt[x]
        else: head[c] = nxt[x]
        if nxt[x] != -1: pre[nxt[x]] = pre[x]    
        pre[x] = nxt[x] = -1
        cnt[x] -= 1
        Add(cnt[x], x)
        if head[mcnt] == -1: mcnt -= 1
    for i in range(1, n+1):
        if cnt[i]:
            Add(cnt[i], i)
            if mcnt < cnt[i]: mcnt = cnt[i]
    if 2*mcnt-1 > n: print(-1); return
    idx, size = 0, n
    p1, p2 = 0, 1
    while p2 < n and a[p2-1] == a[p2]: p2 += 1
    while idx < n and 2*mcnt-1 < size:
        while a[p1] == a[p2] or vis[p1]: p1 += 1
        vis[p1] = 1
        size -= 1
        Remove(cnt[a[p1]], a[p1])
        ans[idx] = p1+1; idx += 1
        if idx == n or 2*mcnt-1 == size: break
        while a[p1] == a[p2] or vis[p2]: p2 += 1
        vis[p2] = 1
        size -= 1
        Remove(cnt[a[p2]], a[p2])
        ans[idx] = p2+1; idx += 1
    maxv = head[mcnt]
    p1 = 0
    while vis[p1]: p1 += 1
    p2 = p1
    while(idx < n):
        while a[p1] != maxv or vis[p1]: p1 += 1
        vis[p1] = 1
        ans[idx] = p1+1; idx += 1
        if idx == n: break
        while a[p2] == maxv or vis[p2]: p2 += 1
        vis[p2] = 1
        ans[idx] = p2+1; idx += 1
    print(*ans)
    return
if __name__ == "__main__":
    main()
```

```c++
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

int main(){
    cin.tie(0); cout.tie(0); ios::sync_with_stdio(false);
    int n; cin >> n;
    vector<int> a(n);
    vector<int> ans(n, 0), cnt(n+1, 0), head(n+1, -1), nxt(n+1, -1), pre(n+1, -1);
    vector<bool> vis(n, 0);
    int mcnt = 0;
    for(int i = 0; i < n; i++){
        cin >> a[i];
        cnt[a[i]]++;
    }
    function<void(int, int)> Add = [&](int c, int x){
        nxt[x] = head[c];
        if(head[c] != -1) pre[head[c]] = x;
        head[c] = x;
    };
    function<void(int, int)> Remove = [&](int c, int x){
        if(pre[x] != -1) nxt[pre[x]] = nxt[x];
        else head[c] = nxt[x];
        if(nxt[x] != -1) pre[nxt[x]] = pre[x];
        pre[x] = nxt[x] = -1;
        cnt[x]--;
        Add(cnt[x], x);
        if(head[mcnt] == -1) mcnt--;
    };
    for(int i = 1; i <= n; i++) if(cnt[i]){
        Add(cnt[i], i);
        if(mcnt < cnt[i]) mcnt = cnt[i];
    }
    if(2*mcnt-1 > n) return !(cout << -1);
    int idx = 0, size = n, p1 = 0, p2 = 1;
    while(p2 < n && a[p2-1] == a[p2]) p2++;
    while(idx < n && 2*mcnt-1 < size){
        while(a[p1] == a[p2] || vis[p1]) p1++;
        vis[p1] = 1;
        size--;
        Remove(cnt[a[p1]], a[p1]);
        ans[idx++] = p1+1;
        if(idx == n || 2*mcnt-1 == size) break;
        while(a[p1] == a[p2] or vis[p2]) p2++;
        vis[p2] = 1;
        size--;
        Remove(cnt[a[p2]], a[p2]);
        ans[idx++] = p2+1;
    }
    int maxv = head[mcnt];
    p1 = 0;
    while(vis[p1]) p1++;
    p2 = p1;
    while(idx < n){
        while(a[p1] != maxv || vis[p1]) p1++;
        vis[p1] = 1;
        ans[idx++] = p1+1;
        if(idx == n) break;
        while(a[p2] == maxv || vis[p2]) p2++;
        vis[p2] = 1;
        ans[idx++] = p2+1;
    }
    for(int i = 0; i < n; i++) cout << ans[i] << ' ';
    return 0;
}
```
