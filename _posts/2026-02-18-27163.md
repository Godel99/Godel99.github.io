[문제 링크](https://www.acmicpc.net/problem/27163) 

## **풀이**
### **DP 설계**
라이프가 1 이상이면 살아남을 수 있고, 오라는 높을수록 생존에 유리해집니다. 이 점을 착안하여 DP를 설계해 줍시다.

* **$dp[i][j]$: $i$번째 공격을 받은 뒤 라이프가 $j$ 이상 남았을 때 남아 있는 최대 오라**

최대 오라 값을 저장하기 위하여 -1로 초기화 해줍니다.

공격을 받을 때마다 오라나 라이프를 잃어버리므로, Bottom-up 방식을 씁시다. 결과적으로 구하고자 하는 값은 $dp[N][1]$일 때가 0 이상이면 ```YES``` -1 이면 ```NO``` 가 됩니다.

### **어떻게 살아남았지?**
위 dp 정의를 통해서 생존 여부를 알 수 있지만, 문제에서 요구하는 살아남은 방법을 알 수 없습니다. 그럼 **역추적**을 위해서 dp 하나를 더 설계합시다.

* **$way[i][j]$: $dp[i][j]$ 의 값을 최대화하기 위해 $i$번째 공격을 맞는 방법 (false이면 라이프, true이면 오라)**

$dp$를 갱신할 때마다 $way$도 함께 갱신해줘서 결과를 역으로 출력해주면 됩니다.

```python
import sys
def print(*args, sep=" ", end="\n"): return sys.stdout.write(sep.join(map(str, args)) + end)
def input(): return sys.stdin.readline().rstrip()

def main():
    n, a, l = map(int, input().split())
    dp = [[-1]*(l+1) for _ in range(n+1)]
    way = [[False]*(l+1) for _ in range(n+1)]
    dam = [0]*(n+1)
    for i in range(1, l+1): dp[0][i] = a
    for i in range(1, n+1):
        x, y = map(int, input().split())
        dam[i] = y
        if x == -1:
            for j in range(l, y, -1):
                if dp[i-1][j] != -1:
                    dp[i][j-y] = dp[i-1][j]
                    way[i][j-y] = True
        elif y == -1:
            for j in range(l, 0, -1):
                if dp[i-1][j] != -1:
                    dp[i][j] = max(0, dp[i-1][j]-x)
                    way[i][j] = False
        else:
            for j in range(1, l+1):
                if dp[i-1][j] >= x:
                    dp[i][j] = dp[i-1][j]-x
                    way[i][j] = False
                if j > y and dp[i][j-y] < dp[i-1][j]:
                    dp[i][j-y] = dp[i-1][j]
                    way[i][j-y] = True
    if dp[n][1] != -1:
        print('YES')
        life = 1
        st = []
        for i in range(n, 0, -1):
            if way[i][life]:
                st.append('L')
                life += dam[i]
            else: st.append('A')
        print(''.join(reversed(st)))
    else: print('NO')
    return 0
if __name__ == '__main__':
    sys.exit(main())
```
```c++
#include<bits/stdc++.h>
using namespace std;
using ll = long long;

int main(){
    cin.tie(0); cout.tie(0); ios::sync_with_stdio(false);
    int n, a, l; cin >> n >> a >> l;
    vector<vector<int>> dp(n+1, vector<int>(l+1, -1));
    vector<vector<bool>> way(n+1, vector<bool>(l+1, false));
    vector<int> dam(n+1, 0);
    for(int i = 1; i <= l; i++) dp[0][i] = a;
    for(int i = 1; i <= n; i++){
        int x, y; cin >> x >> y;
        dam[i] = y;
        if(x == -1){
            for(int j = l; j > y; j--){
                if(dp[i-1][j] != -1){
                    dp[i][j-y] = dp[i-1][j];
                    way[i][j-y] = true;
                }
            }
        }
        else if(y == -1){
            for(int j = l; j > 0; j--){
                if(dp[i-1][j] != -1){
                    dp[i][j] = max(0, dp[i-1][j]-x);
                    way[i][j] = false;
                }
            }
        }
        else{
            for(int j = 1; j <= l; j++){
                if(dp[i-1][j] >= x){
                    dp[i][j] = dp[i-1][j]-x;
                    way[i][j] = false;
                }
                if(j > y && dp[i][j-y] < dp[i-1][j]){
                    dp[i][j-y] = dp[i-1][j];
                    way[i][j-y] = true;
                }
            }
        }
    }
    if(dp[n][1] != -1){
        cout << "YES\n";
        int life = 1;
        stack<char> st;
        for(int i = n; i > 0; i--){
            if(way[i][life]){
                st.push('L');
                life += dam[i];
            }
            else st.push('A');
        }
        while(st.size()){
            cout << st.top();
            st.pop();
        }
    }
    else cout << "NO\n";
    return 0;
}
```

* 시간복잡도: $\mathcal{O}(N \times L)$
* 공간복잡도: $\mathcal{O}(N \times L)$
