## **풀이**
### **$l$로 나눈 나머지가 같은 조각은 전부 맛이 같다**
연속한 구간 $l$의 맛은 항상 같아야 합니다. **원형으로 연속된 구간**이기에 맨 왼쪽 조각을 떼어서 맨 오른쪽 조각에 붙였다고 가정했을 때도 그 맛은 다 같아야 합니다. 결국 **조각을 $l$로 나눈 나머지가 같은 조각들은 전부 맛이 같아야 합니다.**

$i,\ i+1,\ i+2, \dots,\ i+l-1$ 구간에서 왼쪽 조각을 떼어서 오른쪽에 붙였다고 하면, $i+1,\ i+2, \dots,\ i+l-1, \ i+l$ 가 됩니다.

즉, $맛[i]=맛[i+l]$ 입니다. 이걸 반복하면 $맛[1]=맛[1+l]=맛[1+2l]=맛[1+3l]=\cdots$ 같이 $l$로 나눈 나머지가 같은 조각들이 전부 같은 맛을 낸다는 결론을 도출할 수 있습니다.

### **해시(Hash)**

$N,\ x,\ t$가 매우 크므로, 나머지 모음을 인덱스를 통한 접근이 가능한 **해시**를 사용해 빠르게 질의를 해결합시다.

나머지 모음마다 토핑 맛의 개수, 나머지 모음마다 서로 다른 맛이 몇 개 있는지 개수, 몇 번째 조각에 어떤 토핑 맛이 있는지를 모두 해시로 관리해 줍니다.

* $bucket[r][t]$: r(나머지 그룹)에서 맛 t 토핑이 몇 개 있는지 저장
* $bucket\\_cnt[r]$: r 그룹에 서로 다른 맛 종류가 몇 개 있는지 저장
* $mp[x]$: 현재 x번 조각에 어떤 토핑이 있는지 저장

```python
import sys
def print(*args, sep=" ", end="\n"): return sys.stdout.write(sep.join(map(str, args)) + end)
def input(): return sys.stdin.readline().rstrip()

from collections import defaultdict
        
def main():
    n, l, q = map(int, input().split())
    bucket = defaultdict(lambda: defaultdict(int))
    bucket_cnt = defaultdict(int)
    mp = defaultdict(int)
    bad = sum = 0 
    def reset_tp(x):
        if x not in mp: return
        nonlocal sum, bad
        r = x%l
        bucket[r][mp[x]] -= 1
        if bucket[r][mp[x]] == 0:
            bucket_cnt[r] -= 1
            if bucket_cnt[r] == 1: bad -= 1
            sum -= mp[x]
        del mp[x]
    for _ in range(q):
        x = tuple(map(int, input().split()))
        if x[0] == 1:
            r = x[1]%l; t = x[2]
            reset_tp(x[1])
            mp[x[1]] = t
            if bucket[r][t] == 0:
                bucket_cnt[r] += 1
                if bucket_cnt[r] == 2: bad += 1
                sum += t
            bucket[r][t] += 1
        else: reset_tp(x[1])
        if bad: print('NO')
        else: print('YES', sum)
    return 0
if __name__ == '__main__':
    sys.exit(main())
```
```c++
#include<bits/stdc++.h>
using namespace std;
using ll = long long;

ll n, l;
unordered_map<ll, unordered_map<ll, ll>> bucket;
unordered_map<ll, ll> bucket_cnt;
unordered_map<ll, ll> mp;
int bad, q; ll sum;

void reset_tp(ll x){
    if(mp.find(x)==mp.end()) return;
    ll r = x%l;
    bucket[r][mp[x]]--;
    if(!bucket[r][mp[x]]){
        bucket_cnt[r]--;
        if(bucket_cnt[r] == 1) bad--;
        sum -= mp[x];
    }
    mp.erase(x);
}

int main(){
    cin.tie(0); cout.tie(0); ios::sync_with_stdio(false);
    cin >> n >> l >> q;
    while(q--){
        int op; cin >> op;
        if(op == 1){
            ll x, t, r; cin >> x >> t; 
            reset_tp(x);
            mp[x] = t;
            r = x%l;
            if(!bucket[r][t]){
                bucket_cnt[r]++;
                if(bucket_cnt[r] == 2) bad++;
                sum += t;
            }
            bucket[r][t]++;
        }
        else{
            ll x; cin >> x;
            reset_tp(x);
        }
        if(bad) cout << "NO\n";
        else cout << "YES " << sum << '\n';
    }
    return 0;
}
```
