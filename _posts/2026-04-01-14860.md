[문제 링크](https://www.acmicpc.net/problem/14860) 
## **풀이**
$\prod_{i=1}^{N} \prod_{j=1}^{M} \text{gcd}(i, j)$를 구해야 합니다.

### **더블 카운팅**
약수와 관련된 문제입니다. 일일이 최대공약수를 찾아줘도 되지만, $N,\ M$이 커서 시간초과가 납니다. 약수 문제는 배수의 개수를 세는 문제로 바꿔 좀 더 쉽고 빠르게 해결 할 수 있습니다. 이러한 접근 방법을 **더블 카운팅**이라고 합니다.

### **결국은 소수**
모든 자연수는 소수의 곱으로 표현될 수 있고, $\text{GCD}(i, j)$ 역시 마찬가지입니다. 구하고자 하는 것은 $\text{GCD}(i, j)$의 곱이므로 $\text{GCD}(i, j)$의 소인수는 결국 소수의 곱으로 최종적으로 표현할 수 있습니다. 예를 들어, $\text{GCD}(2,2)=2^1$, $\text{GCD}(2,4)=2^1$, $\text{GCD}(4,8)=2^2$과 같이 결국은 **소수의 배수**의 지수 더하기 문제로 바꿔서 생각해 볼 수 있습니다.

$$\prod_{i, j} \text{GCD}(i, j) = 2^{(\text{모든 쌍의 GCD가 가진 2의 개수 합})} \times 3^{(\text{모든 쌍의 GCD가 가진 3의 개수 합})} \times \cdots$$

1. 먼저 $\min(N,\ M)$까지의 소수를 에라토스테네스의 체를 이용해 구해줍니다.
2. 각 소수 $p_i$마다 $\frac{N}{p_i}$와 $\frac{M}{p_i}$를 통해 배수의 개수를 구해줍니다.
3. 배수의 개수를 곱해줘서 지수의 합을 구해주면 최종적으로 $\prod_{i=1}^{N} \prod_{j=1}^{M} \text{gcd}(i, j)$를 구할 수 있습니다.

```python
import sys 
print = lambda *args, sep=" ", end="\n": sys.stdout.write(sep.join(map(str, args)) + end)
input = lambda: sys.stdin.readline().rstrip('\r\n')

MOD = 1_000_000_007

def main():
    n, m = map(int, input().split())
    if n > m: n, m = m, n
    size = n//3+1
    isp = bytearray([1])*size
    isp[0] = 0
    limit = int(n**0.5)
    for i in range(1, limit//3+1):
        if isp[i]:
            p = 3*i+1|1
            d, s, j = p*2, p*p, p*(p+4-2*(i&1))
            isp[s//3::d] = bytearray([0])*((size-1-s//3)//d+1)
            isp[j//3::d] = bytearray([0])*((size-1-j//3)//d+1)
    primes = [2, 3]+[3*i+1|1 for i in range(1, size) if isp[i]]
    ans = 1
    for p in primes:
        pi, ex = p, 0
        while pi <= n:
            ex = (ex+(n//pi)*(m//pi))
            pi *= p
        ans = ans*pow(p, ex%(MOD-1), MOD)%MOD
    print(ans)
    return
if __name__ == '__main__':
    main()
```

```c++
#include<bits/stdc++.h>
using namespace std;
using ll = long long;

const ll MOD = 1'000'000'007;

ll pw(ll n, ll r, ll MOD){
    n %= MOD;
    ll ret = 1;
    while(r){
        if(r&1) ret = ret*n%MOD;
        n = n*n%MOD;
        r >>= 1;
    }
    return ret;
}

int main(){
    cin.tie(0); cout.tie(0); ios::sync_with_stdio(false);
    int n, m; cin >> n >> m;
    if(n > m) swap(n, m);
    int size = n/3+1;
    vector<bool> isp(size, 1);
    isp[0] = 0;
    for(int i = 1; i*i <= size/3; i++){
        if(isp[i]){
            int p = 3*i+1|1;
            int d = p*2;
            int s = p*p;
            int j = p*(p+4-2*(i&1));
            for(int k = s/3; k < size; k += d) isp[k] = 0;
            for(int k = j/3; k < size; k += d) isp[k] = 0;
        }
    }
    vector<int> primes = {2, 3};
    for(int i = 1; i < size; i++) if(isp[i]) primes.push_back(3*i+1|1);
    ll ans = 1;
    for(auto p : primes){
        ll pi = p, ex = 0;
        while(pi <= n){
            ex = (ex+(n/pi)*(m/pi));
            pi *= p;
        }
        ans = ans*pw(p, ex%(MOD-1), MOD)%MOD;
    }
    cout << ans;
    return 0;
}
```
